/* ------------------ 動態計算真實行車時間 (OSRM API) ------------------ */ async function calculateTravelTimes() { const carTimeElem = document.getElementById('car-travel-time'); const busTimeElem = document.getElementById('bus-travel-time'); carTimeElem.innerText = "計算中..."; busTimeElem.innerText = "計算中..."; // 確保查詢時帶有「香港」以利精確定位 const originStr = trafficConfig.origin.includes('香港') ? trafficConfig.origin : `${trafficConfig.origin}, 香港`; const destStr = trafficConfig.destination.includes('香港') ? trafficConfig.destination : `${trafficConfig.destination}, 香港`; try { // 1. 使用 Nominatim 將起點與終點中文地址轉換為經緯度 (Geocoding) const geocodeUrl = (address) => `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(address)}&limit=1`; const [originRes, destRes] = await Promise.all([ fetch(geocodeUrl(originStr)).then(r => r.json()), fetch(geocodeUrl(destStr)).then(r => r.json()) ]); if (!originRes.length || !destRes.length) { throw new Error("無法定位地址"); } const originCoords = `${originRes[0].lon},${originRes[0].lat}`; const destCoords = `${destRes[0].lon},${destRes[0].lat}`; // 2. 呼叫 OSRM 路由 API 計算真實道路駕車距離與時間 const osrmUrl = `https://router.project-osrm.org/route/v1/driving/${originCoords};${destCoords}?overview=false`; const osrmRes = await fetch(osrmUrl).then(r => r.json()); if (osrmRes.code !== 'Ok' || !osrmRes.routes.length) { throw new Error("無法計算路線"); } // OSRM 回傳單位為秒 (duration) const durationSeconds = osrmRes.routes[0].duration; // 取得即時交通快慢係數 (香港運輸署快慢速地圖) let speedFactor = 1.0; try { const speedRes = await fetch('https://rt.data.gov.hk/v1/transport/traffic-speed-map'); const speedData = await speedRes.json(); if (Array.isArray(speedData)) { const congestedRoads = speedData.filter(r => r.TRAFFIC_SPEED_LEVEL === 'S' || r.TRAFFIC_SPEED_LEVEL === 'M'); if (congestedRoads.length > 30) speedFactor = 1.3; else if (congestedRoads.length > 15) speedFactor = 1.15; } } catch(e) { // 若速度 API 失敗則維持 1.0 } // 計算私家車預估時間 (加上即時路況係數) let carMinutes = Math.max(2, Math.round((durationSeconds / 60) * speedFactor)); // 巴士時間估算：加上停站與多繞路時間（長途約 1.35 倍 + 5 分鐘，短途約 1.5 倍 + 3 分鐘） let busMinutes = carMinutes < 10 ? Math.round(carMinutes * 1.5) + 3 : Math.round(carMinutes * 1.35) + 5; // 更新 UI carTimeElem.innerText = `約 ${carMinutes} 分鐘`; busTimeElem.innerText = `約 ${busMinutes} 分鐘`; } catch (e) { console.error("時間計算失敗：", e); carTimeElem.innerText = "無法估算"; busTimeElem.innerText = "無法估算"; } }
